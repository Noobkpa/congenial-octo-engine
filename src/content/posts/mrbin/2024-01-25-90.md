<span style="font-family: impact, sans-serif; font-size: 18pt;">实现代码如下：</span>
<pre class="pure-highlightjs"><code class="">#include&lt;iostream&gt;
#include&lt;time.h&gt;
using namespace std;

typedef struct Node{
	int key;
	struct Node *lchild, *rchild;
}Node;

Node *getnewNode(int key){
	Node *p = new Node;
	p-&gt;key=key;
	p-&gt;lchild=p-&gt;rchild=NULL;
	return p;
}

Node *insert(Node *root,int key){
	if(root==NULL)	return getnewNode(key);
	if(rand()%2)	root-&gt;lchild=insert(root-&gt;lchild,key);
	else			root-&gt;rchild=insert(root-&gt;rchild,key);
	return root;
}

void clear(Node *root)
{
	if(root==NULL)	return;
	clear(root-&gt;lchild);
	clear(root-&gt;rchild);
	free(root);
	return;
}

#define MAX_NODE 10
Node *queue[MAX_NODE+5];
int head,tail;

void bfs(Node *root){
	head=tail=0;
	queue[tail++]=root;
	while(head&lt;tail){
			Node *node=queue[head];
			cout&lt;&lt;endl&lt;&lt;"node: "&lt;&lt;node-&gt;key&lt;&lt;endl;
	if(node-&gt;lchild){
			queue[tail++]=node-&gt;lchild;
			cout&lt;&lt;' '&lt;&lt;node-&gt;key&lt;&lt;"-&gt;"&lt;&lt;node-&gt;lchild-&gt;key&lt;&lt;"(left)"&lt;&lt;endl;
		}
	if(node-&gt;rchild){
			queue[tail++]=node-&gt;rchild;
			cout&lt;&lt;' '&lt;&lt;node-&gt;key&lt;&lt;"-&gt;"&lt;&lt;node-&gt;rchild-&gt;key&lt;&lt;"(right)"&lt;&lt;endl;
		}
		head++;	
	}
	cout&lt;&lt;endl;
	return;
}

int tot=0;
void dfs(Node *root)
{
	if(root==NULL) return;
	int start,end;
	tot+=1;
	start=tot;
	if(root-&gt;lchild)	dfs(root-&gt;lchild);
	if(root-&gt;rchild)	dfs(root-&gt;rchild);
	tot+=1;
	end=tot;
	cout&lt;&lt;root-&gt;key&lt;&lt;" : "&lt;&lt;"["&lt;&lt;start&lt;&lt;','&lt;&lt;end&lt;&lt;"]"&lt;&lt;endl;
	return;
}


int main()
{
	srand(time(0));
	Node *root=NULL;
	for(int i=0;i&lt;MAX_NODE;i++)
	{
		root=insert(root,rand()%100);
	}
	bfs(root);
	dfs(root);
	return 0;
}</code></pre>
